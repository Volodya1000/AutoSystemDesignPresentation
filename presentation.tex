\begin{frame}{Пример BRD: оформление заказа}
  {\small\color{Muted}Фрагмент бизнес-требований интернет-магазина}
  \vspace{0.20cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.54\textwidth}
      \fcolorbox{Line}{Panel}{\parbox[c][4.0cm][c]{0.91\linewidth}{%
        \textbf{Что должна делать система}\par\vspace{0.16cm}
        \begin{itemize}
          \setlength{\itemsep}{0.12cm}
          \item Покупатель оформляет заказ.
          \item Система проверяет наличие товаров.
          \item После оплаты отправляет подтверждение.
          \item Покупатель видит статус заказа.
        \end{itemize}
      }}
    \column{0.43\textwidth}
      \fcolorbox{Line}{Panel}{\parbox[c][4.0cm][c]{0.88\linewidth}{%
        \textbf{Условия работы}\par\vspace{0.16cm}
        \begin{itemize}
          \setlength{\itemsep}{0.16cm}
          \item 120 запросов/с обычно
          \item До 400 запросов/с в пике
          \item Цель: 300 мс для p95
          \item Доступность: 99,9\%
        \end{itemize}
      }}
  \end{columns}
  \vfill
  {\small\color{Muted}Это требования заказчика, а не заранее готовая схема сервисов.}
\end{frame}
