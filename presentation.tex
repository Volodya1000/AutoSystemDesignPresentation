\begin{frame}{Что делает AutoSystemDesign}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \textbf{Проектирование}
      \begin{itemize}
        \item извлекает функциональные и нефункциональные требования из BRD;
        \item строит типизированную C4-модель компонентов;
        \item формирует Sequence-диаграмму для каждого требования;
        \item проектирует OpenAPI 3.1 и Swagger UI.
      \end{itemize}
    \column{0.48\textwidth}
      \textbf{Проверка и анализ}
      \begin{itemize}
        \item проверяет граф, связи, протоколы и трассировку требований;
        \item исправляет структурные ошибки через ограниченный repair loop;
        \item оценивает загрузку, задержки и критический путь;
        \item сохраняет редактируемые и машиночитаемые артефакты.
      \end{itemize}
  \end{columns}
  \vfill
  \muted{LLM предлагает модель; программные валидаторы определяют, можно ли передать её следующей стадии.}
\end{frame}

\begin{frame}{Единый пример: сервис заказов интернет-магазина}
  \begin{columns}[T,onlytextwidth]
    \column{0.55\textwidth}
      \textbf{Функциональные требования}
      \begin{itemize}
        \item \textbf{FR-001}: регистрация заказа по списку товаров с возвратом номера;
        \item \textbf{FR-002}: получение текущего статуса заказа по номеру.
      \end{itemize}
      \vspace{0.20cm}
      \textbf{Граница примера}
      \begin{itemize}
        \item оплата и управление складом не входят в систему;
        \item один BRD используется на всех следующих слайдах.
      \end{itemize}
    \column{0.41\textwidth}
      \textbf{Нагрузка и ограничения}
      \begin{itemize}
        \item 120 запросов/с обычно;
        \item 400 запросов/с в пике;
        \item 80\% чтений, 20\% записей;
        \item p95 ответа API $\leq$ 300 мс;
        \item бюджет $\leq$ 500 USD/месяц.
      \end{itemize}
  \end{columns}
  \vfill
  \muted{Источник: \texttt{examples/requirements/online\_store\_brd.md}.}
\end{frame}

\begin{frame}{Repair loop: исправление структурного ответа}
  \begin{enumerate}
    \item LLM формирует JSON по схеме доменной модели.
    \item Ответ разбирается и передаётся доменным валидаторам.
    \item Ошибка JSON или нарушенное правило преобразуется в конкретную диагностику.
    \item Диагностика добавляется в следующую попытку генерации.
    \item После успешной проверки создаются артефакты; число попыток ограничено.
  \end{enumerate}
  \vspace{0.25cm}
  \begin{center}
    \begin{tikzpicture}[node distance=0.55cm,
      box/.style={draw=Line,fill=Panel,rounded corners=2pt,align=center,minimum height=0.8cm,text width=2.4cm},
      arrow/.style={draw=Accent,thick,-{Stealth[length=2mm]}}]
      \node[box] (llm) {LLM JSON};
      \node[box,right=of llm] (parse) {Pydantic};
      \node[box,right=of parse] (valid) {Валидаторы};
      \node[box,right=of valid] (out) {Артефакты};
      \draw[arrow] (llm)--(parse); \draw[arrow] (parse)--(valid); \draw[arrow] (valid)--(out);
      \draw[arrow] (valid.south) |- ++(0,-0.55) -| node[pos=.5,below,font=\scriptsize]{диагностика} (llm.south);
    \end{tikzpicture}
  \end{center}
\end{frame}

\begin{frame}{Sequence-диаграммы связаны с C4}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \textbf{Правила генерации}
      \begin{itemize}
        \item отдельный сценарий на каждое FR;
        \item участники ссылаются на роли и узлы C4;
        \item переходы должны существовать в архитектурном графе;
        \item протокол влияет на вид сообщения;
        \item ветвления описываются типизированными блоками.
      \end{itemize}
    \column{0.48\textwidth}
      \textbf{Проверки}
      \begin{itemize}
        \item все объявленные линии жизни используются;
        \item инициатор и бизнес-роль согласованы;
        \item события и уведомления асинхронны;
        \item идентификатор требования сохраняется для API.
      \end{itemize}
  \end{columns}
  \vfill
  \muted{Sequence не является независимой картинкой: она проверяется относительно уже принятой C4-модели.}
\end{frame}

\begin{frame}{Как рассчитывается производительность}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \textbf{Распространение потока}
      \begin{itemize}
        \item пиковый RPS проходит по синхронным связям;
        \item ветви учитывают вероятность и routing mode;
        \item cache hit rate разделяет быстрый путь и обращение к БД.
      \end{itemize}
    \column{0.48\textwidth}
      \textbf{Расчёт по узлам}
      \begin{itemize}
        \item ёмкость = реплики × каналы / время обслуживания;
        \item вычисляются utilization, ожидание и средний ответ;
        \item самый длинный синхронный путь выделяется отдельно.
      \end{itemize}
  \end{columns}
  \vspace{0.25cm}
  \fcolorbox{Line}{Panel}{\parbox{0.94\textwidth}{\small
    Для быстрого запуска параметры берутся из C4-кандидата. Если есть измерения, \texttt{--profiles} подменяет их явным воспроизводимым каталогом.}}
\end{frame}
