\begin{frame}{Какую задачу решает инструмент}
  \vspace{0.30cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      {\large\bfseries Проблема}\par\vspace{0.23cm}
      \begin{itemize}
        \setlength{\itemsep}{0.29cm}
        \item Архитектор тратит время на ручной перевод требований в диаграммы компонентов.
        \item Диаграммы компонентов, диаграммы последовательностей и API необходимо согласовывать между собой.
      \end{itemize}
    \column{0.47\textwidth}
      {\large\bfseries Решение}\par\vspace{0.23cm}
      \begin{itemize}
        \setlength{\itemsep}{0.22cm}
        \item LLM формирует структурированную модель компонентов и их взаимодействий.
        \item Детерминированные валидаторы проверяют типы, связи и ограничения модели.
        \item На её основе строятся артефакты (C4, диаграммы последовательностей, OpenAPI, отчёт производительности).
      \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Генерация архитектурной модели и исправление ошибок}
  \vspace{0.20cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      {\large\bfseries Генерация}\par\vspace{0.20cm}
      \begin{itemize}
        \setlength{\itemsep}{0.25cm}
        \item LLM возвращает структурированное описание компонентов, связей и взаимодействий.
        \item Из проверенной модели программно строятся диаграммы и спецификации.
      \end{itemize}
    \column{0.48\textwidth}
      {\large\bfseries Проверка и повторная генерация}\par\vspace{0.20cm}
      \begin{itemize}
        \setlength{\itemsep}{0.22cm}
        \item Валидаторы проверяют обязательные поля, типы и связи.
        \item При ошибке LLM получает перечень нарушений и исправляет модель.
        \item Количество повторных попыток ограничено.
      \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Модель компонентов: зачем нужны типы и параметры}
  \vspace{0.29cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      {\large\bfseries Типы → проверка связей}\par\vspace{0.18cm}
      \begin{itemize}
        \setlength{\itemsep}{0.25cm}
        \item Узел помечается типом: клиент, сервис, база данных, кэш, брокер.
        \item Тип определяет, какие связи между узлами допустимы.
        \item Например: клиент → API → база данных.
      \end{itemize}
    \column{0.47\textwidth}
      {\large\bfseries Параметры → расчёты}\par\vspace{0.18cm}
      \begin{itemize}
        \setlength{\itemsep}{0.25cm}
        \item Число реплик, время обработки запроса и размер очереди.
        \item По ним оцениваются пропускная способность, загрузка и ожидание.
      \end{itemize}
  \end{columns}
  \vspace{0.28cm}
  \fcolorbox{Line}{Panel}{\parbox{0.96\linewidth}{%
    \small Одна модель используется для построения диаграммы компонентов
    и проверки нагрузки на архитектуру.
  }}
\end{frame}

\begin{frame}{Проверка диаграмм последовательностей}
  \vspace{0.35cm}
  {\large\bfseries Проверяемые инварианты}\par\vspace{0.25cm}
  \begin{enumerate}
    \setlength{\itemsep}{0.40cm}
    \item Идентификатор требования на диаграмме должен совпадать
      с требованием, для которого она построена.
    \item Каждый компонент на диаграмме должен ссылаться
      на существующий узел графа архитектуры.
    \item Сообщения-события и уведомления должны иметь
      асинхронный режим доставки.
  \end{enumerate}
\end{frame}
